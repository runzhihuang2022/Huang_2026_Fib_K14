import stereo as st
import warnings
import os

# 屏蔽无关警告
warnings.filterwarnings('ignore')

# ================= 配置路径 =================
# 对应你的 st_rds_dir
st_rds_dir = r"data/external/local_728c8c962d2d"  
# 结果输出目录
output_dir = r"data/external/local_c9ce0f6697a4"
os.makedirs(output_dir, exist_ok=True)

# ================= 获取文件列表 =================
# 相当于 R 的 list.files(pattern = "\\.rds$")
all_st_files = [f for f in os.listdir(st_rds_dir) if f.endswith('.rds')]

print(f"📂 找到 {len(all_st_files)} 个 RDS 文件，准备开始转换...")

# ================= 批处理循环 =================
for file_name in all_st_files:
    # 提取样本名（去除 .rds 后缀）
    sample_id = file_name.rsplit('.', 1)[0]
    input_path = os.path.join(st_rds_dir, file_name)
    output_path = os.path.join(output_dir, f"{sample_id}.h5ad")
    
    print(f"🚀 正在处理: {sample_id}")
    
    try:
        # 1. 读取 Seurat 对象
        # StereoPy 会尝试通过 rpy2 调用 R 来解析 RDS
        data = st.io.read_rds(input_path)
        
        # 2. 执行质控
        data.tl.cal_qc()
        
        # 3. 保存原始计数快照
        data.tl.raw_checkpoint()
        
        # 4. 转换并保存为 h5ad
        # 如果后续要进 Scanpy 分析，维持 'scanpy'；若回 Seurat，改为 'seurat'
        st.io.stereo_to_anndata(
            data, 
            flavor='scanpy', 
            output=output_path
        )
        
        print(f"✅ {sample_id} 转换成功")
        
    except Exception as e:
        print(f"❌ {sample_id} 处理失败: {str(e)}")

print("\n✨ 所有样本处理并导出完成！")

import stereo as st
import os
import warnings
import rpy2.robjects as robjects
from rpy2.robjects import r
import anndata2ri

# 屏蔽无关警告
warnings.filterwarnings('ignore')

# ================= 配置路径 =================
# 对应你的 st_rds_dir
st_rds_dir = r"data/external/local_4d178e80268f"  
# 结果输出目录
output_dir = r"data/external/local_4d178e80268f"
os.makedirs(output_dir, exist_ok=True)
all_st_files = ['BW15D_1_D1_SDSDB_19dpb.rds', 'BW15D_C6_SDSDB_19dpb.rds']

# ================= 转换循环 =================
for file_name in all_st_files:
    sample_id = file_name.replace('.rds', '')
    input_path = os.path.join(st_rds_dir, file_name)
    output_path = os.path.join(output_dir, f"{sample_id}.h5ad")
    
    print(f"🚀 正在处理样本: {sample_id}")
    
    try:
        # 1. 使用 R 引擎直接读取 RDS
        # 这一步会将 R 里的 Seurat 对象加载到 Python 内存中
        robjects.r(f'library(Seurat)')
        robjects.r(f'seurat_obj <- readRDS("{input_path}")')
        
        # 2. 转换为 AnnData 对象 (通过 anndata2ri)
        # 注意：这里我们使用 anndata2ri 将 R 的 Seurat 对象转为 Python 可操作的 adata
        adata = robjects.r('as.SingleCellExperiment(seurat_obj)')
        
        # 3. 将其封装进 StereoData 以便使用 StereoPy 的质控功能
        # 如果你只需要 h5ad，可以直接 adata.write_h5ad()
        data = st.io.anndata_to_stereo(adata)
        
        # 4. 执行 StereoPy 质控流程
        data.tl.cal_qc()
        data.tl.raw_checkpoint()
        
        # 5. 导出为最终的 h5ad
        st.io.stereo_to_anndata(data, flavor='scanpy', output=output_path)
        
        print(f"✅ {sample_id} 转换成功 -> {output_path}")
        
    except Exception as e:
        print(f"❌ {sample_id} 转换失败: {str(e)}")

print("\n✨ 任务全部完成！")

# ==========================================
# 1. 严格提取 16 个核心样本并取并集合并
# ==========================================
target_files = [
    "1mph_A03699G6.SCT.h5ad", "6mph_A02988A2.SCT.h5ad", "9mph_A03091C5.SCT.h5ad",
    "BW13_1_B1_DSDB_12dpb.h5ad", "BW13_A3_DSDB_12dpb.h5ad",

    "Y01066D1.adjusted.cellbin.h5ad", "Y01066D2.adjusted.cellbin.h5ad"
]

adata_list = []
sample_names = []
print(f"🎯 锁定指定的 {len(target_files)} 个核心样本，开始循环读取...")

for file_name in target_files:
    path = os.path.join(data_dir, file_name)
    if not os.path.exists(path):
        print(f"⚠️ 找不到文件: {path}")
        continue
    s_name = file_name.replace(".h5ad", "")
    try:
        temp_ad = sc.read_h5ad(path)
        if isinstance(temp_ad.X, np.ndarray): temp_ad.X = sp.csr_matrix(temp_ad.X)
        elif not sp.isspmatrix_csr(temp_ad.X): temp_ad.X = temp_ad.X.tocsr()
            
        temp_ad.uns.clear()
        keys_to_keep = ['spatial']
        for k in list(temp_ad.obsm.keys()):
            if k not in keys_to_keep: del temp_ad.obsm[k]
                
        temp_ad.var_names_make_unique()
        temp_ad.obs_names = s_name + "_" + temp_ad.obs_names
        adata_list.append(temp_ad)
        sample_names.append(s_name)
        print(f"  ✅ 载入: {s_name} (细胞数: {temp_ad.n_obs})")
    except Exception as e: 
        print(f"  ⚠️ 读取 {file_name} 失败: {e}")
    gc.collect()

print("\n🔗 正在采用 outer join 取并集合并 (保留所有出现过的基因，缺失用 0 填充)...")
adata = ad.concat(adata_list, join='outer', label='sample_batch_new', keys=sample_names, index_unique=None, fill_value=0)
del adata_list
gc.collect()
print(f"✨ 合并完毕！当前总矩阵大小：{adata.n_obs:,} 个 Spot，{adata.n_vars:,} 个基因。")

# ==========================================
# 2. 数据标准化与【全基因】备份
# ==========================================
# 1. 核心修复：剔除没有任何表达量的幽灵 Spot (防止分母为0导致 NaN)
print(f"过滤前 Spot 数量: {adata.n_obs:,}")
sc.pp.filter_cells(adata, min_counts=1)
# 顺手把没有任何表达的死基因也剔除，极大地降低计算负担
sc.pp.filter_genes(adata, min_counts=1) 
print(f"过滤后有效 Spot 数量: {adata.n_obs:,}")

# ==========================================
# 2. 重新执行标准化与【全基因】备份
# ==========================================
print("\n⚖️ 正在重新执行标准化...")
sc.pp.normalize_total(adata, target_sum=1e4)
sc.pp.log1p(adata)

print("💾 正在将【全量基因】备份到 adata.raw 中...")
adata.raw = adata

# ==========================================
# 3. 高变基因、防溢出 Scale 与 PCA
# ==========================================
print("🧬 正在计算高变基因...")
sc.pp.highly_variable_genes(adata, n_top_genes=3000, flavor='seurat')

print("⚖️ 正在提取高变子集执行 Scale...")
adata_hvg = adata[:, adata.var['highly_variable']].copy()
sc.pp.scale(adata_hvg, max_value=10)

print("📊 正在计算 PCA...")
sc.tl.pca(adata_hvg, svd_solver='arpack')

# 将健康的 PCA 坐标移交给大矩阵
adata.obsm['X_pca'] = adata_hvg.obsm['X_pca'].copy()
del adata_hvg
gc.collect()

# 👑 核心护航操作：强行斩杀 PCA 矩阵中的所有 NaN / Inf 毒药！
print("🛡️ 正在扫描并清除 PCA 矩阵中的 NaN (非数字)...")
if np.isnan(adata.obsm['X_pca']).any() or np.isinf(adata.obsm['X_pca']).any():
    print("  ⚠️ 警告：检测到 NaN/Inf！这正是导致 Harmony 崩溃的元凶！正在强行抹平修正...")
    # 将所有的 NaN 和无穷大全部替换为 0.0 (相当于把这些点归置到 PCA 坐标原点)
    adata.obsm['X_pca'] = np.nan_to_num(adata.obsm['X_pca'], nan=0.0, posinf=0.0, neginf=0.0)
    print("  ✅ 毒药清除完毕！")
else:
    print("  ✅ PCA 矩阵健康，无 NaN！")

# ==========================================
# 4. 重新启动 Harmony 算法
# ==========================================
print("\n⚖️ 正在启动 Harmony 算法消除批次效应 (此次已免除溢出风险)...")
# 强制设定为类别变量，提升底层运算的稳定性
adata.obs['sample_batch_new'] = adata.obs['sample_batch_new'].astype('category')
sc.external.pp.harmony_integrate(adata, key='sample_batch_new', max_iter_harmony=20)
print("✅ Harmony 校正坐标已成功生成在 adata.obsm['X_pca_harmony']！")

# ==========================================
# 5. 图聚类网络构建与 Leiden 聚类
# ==========================================
print("\n🕸️ 正在基于 Harmony 空间构建细胞邻接图...")
sc.pp.neighbors(adata, use_rep='X_pca_harmony', n_neighbors=15)

print("🧬 正在执行 Leiden 聚类...")
sc.tl.leiden(adata, resolution=1.5, key_added='leiden_harmony')

# 提取固定的颜色映射表
cluster_colors = adata.uns.get('leiden_harmony_colors', None)
print(f"✅ 聚类完成！共划分出 {len(adata.obs['leiden_harmony'].unique())} 个细胞亚群。")

# ==========================================
# 6. 单样本逐级空间可视化
# ==========================================
current_time = datetime.now().strftime("%Y%m%d_%H%M")
pdf_path = os.path.join(out_dir, f"Spatial_Clusters_Harmony_{current_time}.pdf")
print(f"\n🎨 正在绘制每个样本的聚类空间分布图，并保存至: {pdf_path}")

spot_size = 3 

with PdfPages(pdf_path) as pdf:
    for sample in sample_names:
        print(f"  -> 正在渲染: {sample}")
        fig, ax = plt.subplots(figsize=(10, 10))
        
        adata_sub = adata[adata.obs['sample_batch_new'] == sample]

            adata_sub, 
            basis='spatial', 
            color='leiden_harmony', 
            palette=cluster_colors, 
            size=spot_size, 
            frameon=False, 
            show=False, 
            ax=ax, 
            title=f"Sample: {sample} \n(Harmony Integrated Clusters)"
        )
        
        ax.set_aspect('equal', adjustable='datalim')
        if len(ax.collections) > 0: 
            ax.collections[0].set_rasterized(True)
            
        if ax.get_legend() is not None:
            ax.legend(loc='center left', bbox_to_anchor=(1.05, 0.5), frameon=False, ncol=2)
            
        plt.tight_layout()
        pdf.savefig(fig, facecolor='black', dpi=150, bbox_inches='tight')
        plt.close(fig)

print("\n✅ 所有时空图谱出图完毕！")

# ==========================================
# 7. 保存最终成果
# ==========================================
save_path = os.path.join(data_dir, "combined_16samples_Harmony_Clustered.h5ad")
print(f"💾 正在保存最终大矩阵 (包含 .raw 与 Harmony 聚类结果) ...")

print(f"✨ 大功告成！数据已安全存至: {save_path}")

for cell_type, config in annotation_config.items():
    valid_pairs = list(zip(config['markers'], config['cutoffs']))
    mask = np.ones((adata.n_obs,), dtype=bool)
    score_sum = np.zeros((adata.n_obs,))
    
    for g, cut in valid_pairs:
        matrix_g = adata.raw[:, g].X if use_raw else adata[:, g].X
        expr = matrix_g.toarray().flatten() if sp.issparse(matrix_g) else np.asarray(matrix_g).flatten()
        expr = np.nan_to_num(expr, nan=0.0) 
        mask &= (expr > cut)
        score_sum += expr
        
    df_masks[cell_type] = mask
    df_scores[cell_type] = score_sum / len(valid_pairs)

# 3. 强制分配：第一步，基于 base_cells 寻找得分最高的类型 (不设 0 阈值)

        low_signal_mask = df_scores.loc[idx, base_cells].max(axis=1) < 0.1
        adata.obs.loc[idx.index[idx & low_signal_mask], 'sub_labels'] = most_frequent

def create_master_handles(cell_list, all_data):
    handles, labels = [], []
    all_present_cells = all_data.obs['sub_labels'].unique()
    for c in cell_list:
        if c in all_present_cells:
            handles.append(Line2D([0], [0], marker='o', color='w', markerfacecolor=color_palette[c], markersize=12))
            labels.append(c)
    return handles, labels

    mask = np.ones((adata.n_obs,), dtype=bool)
    score_sum = np.zeros((adata.n_obs,))
    
    for g, cut in valid_pairs:
        matrix_g = adata.raw[:, g].X if use_raw else adata[:, g].X
        expr = matrix_g.toarray().flatten() if sp.issparse(matrix_g) else np.asarray(matrix_g).flatten()
        mask &= (expr > cut)
        score_sum += expr

    df_scores[cell_type] = score_sum / len(valid_pairs)
